# Phase 8.5 — Multi-Tenant Foundation Implementation Plan (v2.0 — Production-Safe)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Introduce row-level multi-tenancy into Koinonia — adding `organization_id` isolation to all domain tables, enabling the Better Auth Organizations plugin, building the TenantMiddleware + BaseRepository + PermissionResolver, and shipping the org creation/switching UI.

**Architecture:** Row-level isolation via `organization_id` (TEXT type, matching Better Auth PKs) on all tenant tables; FK compostas prevent cross-tenant references at DB level; `activeOrganizationId` sourced exclusively from the server-side Better Auth session; `PermissionResolver` is the sole authorization function; `BaseRepository` enforces org isolation via `whereOrg()` / `withOrg()` helpers that cannot be bypassed.

**Tech Stack:** PostgreSQL (Drizzle ORM + raw SQL for FK compostas and partial indexes), Fastify, Better Auth v1.5.6 + Organizations plugin, React + TanStack Query v5, TypeScript.

---

## ⚠️ Diff Summary (v1.0 → v2.0)

| # | Change | Reason |
|---|---|---|
| 1 | **Reordered:** Better Auth tables created in Tasks 1–2, BEFORE domain migrations (Tasks 4–8) | Organization table must exist before backfill can reference it |
| 2 | **Replaced:** `UUID` → `TEXT` for all `organization_id` columns | Better Auth uses `text` PKs; UUID FK references would fail |
| 3 | **Fixed:** TenantMiddleware uses `request.headers` (full, not just bearer) for session extraction | Better Auth handles cookie/bearer transparently when given full headers |
| 4 | **Fixed:** BaseRepository exposes `whereOrg(table)` and `withOrg(data)` helpers — not just stores orgId | Storing orgId without enforcing its use is a false sense of security |
| 5 | **Expanded:** FK compostas now also cover `eventos → locais` | Cross-org local assignment was unprotected at DB level |
| 6 | **Fixed:** Organization creation atomically updates member role to `PRESIDENTE` | Better Auth defaults to `owner`; must be normalized before returning |
| 7 | **Fixed:** OrgSwitcher uses `queryClient.clear()` + `navigate()` — removed `window.location.reload()` | Hard reload loses React state; `clear()` + navigate is sufficient |

```typescript
// apps/api/src/scripts/backfill-default-org.ts
// Idempotent: safe to run multiple times.
import 'dotenv/config'
import { db } from '../db'
import { sql } from 'drizzle-orm'

// Use a deterministic text ID (matching Better Auth text PK format)
const DEFAULT_ORG_ID = 'default-org-koinonia-seed'
const DEFAULT_ORG_NAME = 'Igreja Padrão'
const DEFAULT_ORG_SLUG = 'igreja-padrao'

async function backfill() {
  console.log('Starting backfill...')

  // 1. Create default org in the organization table (created by Better Auth plugin)
  await db.execute(sql`
    INSERT INTO organization (id, name, slug, created_at, updated_at)
    VALUES (
      ${DEFAULT_ORG_ID},
      ${DEFAULT_ORG_NAME},
      ${DEFAULT_ORG_SLUG},
      now(),
      now()
    )
    ON CONFLICT (id) DO NOTHING
  `)
  console.log(`  organization: seeded (id=${DEFAULT_ORG_ID})`)

  // 2. Assign default org to all rows where organization_id is null
  const tables = ['pessoas', 'eventos', 'inscricoes', 'locais'] as const
  for (const table of tables) {
    const result = await db.execute(sql.raw(`
      UPDATE "${table}"
      SET organization_id = '${DEFAULT_ORG_ID}'
      WHERE organization_id IS NULL
    `))
    console.log(`  ${table}: updated ${(result as any).rowCount ?? '?'} rows`)
  }

  console.log('Backfill complete.')
  process.exit(0)
}

backfill().catch((err) => {
  console.error('Backfill failed:', err)
  process.exit(1)
})
```

-- 0008_tenant_step2_not_null.sql
-- Type: TEXT FK references organization(id) — type-safe because both are text.
-- Requires: backfill completed (no NULLs in organization_id columns).

-- Make organization_id NOT NULL
ALTER TABLE "pessoas"    ALTER COLUMN "organization_id" SET NOT NULL;
ALTER TABLE "eventos"    ALTER COLUMN "organization_id" SET NOT NULL;
ALTER TABLE "inscricoes" ALTER COLUMN "organization_id" SET NOT NULL;
ALTER TABLE "locais"     ALTER COLUMN "organization_id" SET NOT NULL;

-- Add FK references to organization table (text → text: no type mismatch)
ALTER TABLE "pessoas"    ADD CONSTRAINT "pessoas_org_fk"    FOREIGN KEY ("organization_id") REFERENCES "organization"("id");
ALTER TABLE "eventos"    ADD CONSTRAINT "eventos_org_fk"    FOREIGN KEY ("organization_id") REFERENCES "organization"("id");
ALTER TABLE "inscricoes" ADD CONSTRAINT "inscricoes_org_fk" FOREIGN KEY ("organization_id") REFERENCES "organization"("id");
ALTER TABLE "locais"     ADD CONSTRAINT "locais_org_fk"     FOREIGN KEY ("organization_id") REFERENCES "organization"("id");

-- Unique composite indexes (prerequisite for FK compostas in step 3)
CREATE UNIQUE INDEX "eventos_org_id_uidx"   ON "eventos"   ("organization_id", "id");
CREATE UNIQUE INDEX "pessoas_org_id_uidx"   ON "pessoas"   ("organization_id", "id");
CREATE UNIQUE INDEX "locais_org_id_uidx"    ON "locais"    ("organization_id", "id");
CREATE UNIQUE INDEX "inscricoes_org_id_uidx" ON "inscricoes" ("organization_id", "id");

-- Performance indexes
CREATE INDEX "pessoas_org_lider_idx"      ON "pessoas"   ("organization_id", "lider_pessoa_id");
CREATE INDEX "eventos_org_status_idx"     ON "eventos"   ("organization_id", "status");
CREATE INDEX "inscricoes_org_evento_idx"  ON "inscricoes" ("organization_id", "evento_id");
CREATE INDEX "inscricoes_org_pessoa_idx"  ON "inscricoes" ("organization_id", "pessoa_id");

// apps/api/src/middleware/tenant.ts
//
// DESIGN: Better Auth's auth.api.getSession() accepts raw headers and handles
// both cookie-based and bearer-token sessions transparently. We pass the full
// request.headers object — do NOT extract the token manually.
//
// The session field path (from spike in Task 3) is:
//   session.session.activeOrganizationId
// (Better Auth wraps the DB session in a .session property)

import fp from 'fastify-plugin'
import { FastifyRequest, FastifyReply, FastifyPluginAsync } from 'fastify'
import { and, eq } from 'drizzle-orm'
import { auth } from '../config/auth'
import { db } from '../db'
import { member } from '../db/schema'
import { TenantContext, OrgRole } from '../lib/tenant/types'

declare module 'fastify' {
  interface FastifyRequest {
    tenantCtx?: TenantContext
  }
}

// Header names that must never be used as org source — log and ignore
const SUSPICIOUS_ORG_HEADERS = ['x-org-id', 'x-organization-id', 'organization-id']

const tenantPlugin: FastifyPluginAsync = async (fastify) => {
  fastify.addHook('preHandler', async (request: FastifyRequest, reply: FastifyReply) => {
    // Audit: log attempts to inject org via header
    for (const h of SUSPICIOUS_ORG_HEADERS) {
      if (request.headers[h]) {
        request.log.warn(
          { event: 'suspicious_tenant_header', header: h, ip: request.ip },
          'Client attempted to set org via header — ignored',
        )
      }
    }

    // Audit: log attempts to inject org via body
    const body = request.body as Record<string, unknown> | undefined
    if (body && 'organization_id' in body) {
      request.log.warn(
        { event: 'suspicious_tenant_body', ip: request.ip },
        'Client sent organization_id in body — ignored for tenant resolution',
      )
    }

    // Pass ALL headers — Better Auth resolves cookie or bearer automatically
    let sessionResult: Awaited<ReturnType<typeof auth.api.getSession>> | null = null
    try {
      sessionResult = await auth.api.getSession({ headers: request.headers as any })
    } catch {
      // No valid session — request proceeds without tenant context
      return
    }

    if (!sessionResult) return

    // Adjust field path based on spike findings (Task 3).
    // Better Auth v1.5.x: session data is in sessionResult.session
    const activeOrganizationId = sessionResult.session?.activeOrganizationId
    if (!activeOrganizationId) return // user authenticated but no active org

    const userId = sessionResult.user.id

    const [membership] = await db
      .select({ role: member.role })
      .from(member)
      .where(
        and(
          eq(member.userId, userId),
          eq(member.organizationId, activeOrganizationId),
        ),
      )
      .limit(1)

    if (!membership) {
      // User has a stale activeOrganizationId but is no longer a member
      request.log.warn(
        { event: 'stale_active_org', userId, activeOrganizationId },
        'User has active org but is not a member — clearing context',
      )
      return
    }

    request.tenantCtx = {
      orgId: activeOrganizationId,
      userId,
      userRole: membership.role as OrgRole,
    }
  })
}

export const tenantMiddleware = fp(tenantPlugin)

// Call this in route handlers that require an active org.
// Sends 401 and throws to stop handler execution if ctx is absent.
export function requireTenantCtx(request: FastifyRequest, reply: FastifyReply): TenantContext {
  if (!request.tenantCtx) {
    reply.code(401).send({
      error: 'No active organization. Call POST /auth/organizations/set-active first.',
    })
    // Throwing prevents the rest of the handler from running
    throw new Error('tenant ctx required')
  }
  return request.tenantCtx
}
```

// apps/api/src/lib/tenant/base-repository.ts
import { eq } from 'drizzle-orm'
import { MissingTenantContextError } from './errors'
import { TenantContext } from './types'
import { Database } from '../../db'

export abstract class BaseRepository {
  protected readonly orgId: string

  constructor(
    protected readonly db: Database,
    ctx: TenantContext,
  ) {
    if (!ctx?.orgId) throw new MissingTenantContextError()
    this.orgId = ctx.orgId
  }

  // Use in WHERE clauses to guarantee org scoping.
  // Example: .where(and(this.whereOrg(pessoas), eq(pessoas.id, id)))
  protected whereOrg(table: { organization_id: any }) {
    return eq(table.organization_id, this.orgId)
  }

  // Use in INSERT values to guarantee organization_id is always injected.
  // Example: this.db.insert(table).values(this.withOrg({ nome: 'X' }))
  protected withOrg<T extends Record<string, unknown>>(data: T): T & { organization_id: string } {
    return { ...data, organization_id: this.orgId }
  }
}

// apps/api/src/modules/pessoas/repositories/PessoasRepository.ts
import { and, eq, ilike, isNull, sql } from 'drizzle-orm'
import { Database } from '../../../db'
import { pessoas } from '../../../db/schema'
import { BaseRepository } from '../../../lib/tenant/base-repository'
import { TenantContext } from '../../../lib/tenant/types'

export class PessoasRepository extends BaseRepository {
  constructor(db: Database, ctx: TenantContext) { super(db, ctx) }

  async list(opts: { q?: string; page: number; pageSize: number }) {
    const { q, page, pageSize } = opts
    const offset = (page - 1) * pageSize
    const conditions = [this.whereOrg(pessoas), isNull(pessoas.deleted_at)]
    if (q) conditions.push(ilike(pessoas.nome, `%${q}%`))
    const whereClause = and(...conditions)

    const [data, all] = await Promise.all([
      this.db.select().from(pessoas).where(whereClause).limit(pageSize).offset(offset).orderBy(pessoas.nome),
      this.db.select({ id: pessoas.id }).from(pessoas).where(whereClause),
    ])
    return { data, pagination: { page, pageSize, total: all.length, totalPages: Math.ceil(all.length / pageSize) } }
  }

  async findById(id: string) {
    return this.db.query.pessoas.findFirst({
      where: and(this.whereOrg(pessoas), eq(pessoas.id, id)),
    })
  }

  async create(data: Omit<typeof pessoas.$inferInsert, 'organization_id'>) {
    const [row] = await this.db.insert(pessoas).values(this.withOrg(data)).returning()
    return row
  }

  async update(id: string, data: Partial<typeof pessoas.$inferInsert>) {
    const [row] = await this.db
      .update(pessoas)
      .set({ ...data, updated_at: new Date() })
      .where(and(this.whereOrg(pessoas), eq(pessoas.id, id)))
      .returning()
    return row
  }

  async softDelete(id: string) {
    const [row] = await this.db
      .update(pessoas)
      .set({ deleted_at: new Date() })
      .where(and(this.whereOrg(pessoas), eq(pessoas.id, id)))
      .returning()
    return row
  }

  // CTE recursiva: returns all pessoas in the subtree rooted at pessoaId (inclusive)
  async findSubtree(pessoaId: string, maxDepth = 10): Promise<typeof pessoas.$inferSelect[]> {
    const rows = await this.db.execute(sql`
      WITH RECURSIVE rede AS (
        SELECT p.id, p.lider_pessoa_id, 1 AS depth
        FROM pessoas p
        WHERE p.id = ${pessoaId}
          AND p.organization_id = ${this.orgId}
          AND p.deleted_at IS NULL

        UNION ALL

        SELECT p.id, p.lider_pessoa_id, r.depth + 1
        FROM pessoas p
        INNER JOIN rede r ON p.lider_pessoa_id = r.id
        WHERE p.organization_id = ${this.orgId}
          AND p.deleted_at IS NULL
          AND r.depth < ${maxDepth}
      )
      SELECT p.*
      FROM pessoas p
      WHERE p.id IN (SELECT id FROM rede)
        AND p.organization_id = ${this.orgId}
    `)
    return rows as typeof pessoas.$inferSelect[]
  }

  async findDirectChildren(pessoaId: string) {
    return this.db
      .select()
      .from(pessoas)
      .where(and(this.whereOrg(pessoas), eq(pessoas.lider_pessoa_id, pessoaId), isNull(pessoas.deleted_at)))
  }
}

// apps/api/src/modules/organizations/usecases/TransferPresidencyUseCase.ts
import { sql } from 'drizzle-orm'
import { Database } from '../../../db'
import { TenantContext, Operation } from '../../../lib/tenant/types'
import { TenantForbiddenError } from '../../../lib/tenant/errors'
import { canPerform } from '../../../lib/tenant/permission-resolver'

export class TransferPresidencyUseCase {
  constructor(private db: Database, private ctx: TenantContext) {}

  async execute(newPresidentUserId: string): Promise<void> {
    if (!canPerform(this.ctx.userRole, Operation.TRANSFER_PRESIDENCY)) {
      throw new TenantForbiddenError(Operation.TRANSFER_PRESIDENCY)
    }
    if (this.ctx.userId === newPresidentUserId) {
      throw new Error('Cannot transfer presidency to yourself')
    }

    const { orgId, userId: currentPresidentId } = this.ctx

    await this.db.transaction(async (tx) => {
      // Step 1: demote current president (atomic — must succeed first)
      await tx.execute(sql`
        UPDATE member
        SET role = 'PASTOR_PRINCIPAL'
        WHERE organization_id = ${orgId}
          AND user_id = ${currentPresidentId}
          AND role = 'PRESIDENTE'
      `)

      // Step 2: promote new president
      const result = await tx.execute(sql`
        UPDATE member
        SET role = 'PRESIDENTE'
        WHERE organization_id = ${orgId}
          AND user_id = ${newPresidentUserId}
      `)

      if ((result as any).rowCount === 0) {
        throw new Error('Target user is not a member of this organization')
      }
      // The partial unique index one_president_per_org enforces there is now
      // exactly one PRESIDENTE — if both updates create two, the index rolls back.
    })
  }
}

// DESIGN: Better Auth's organization.create() sets the creator's member role to 'owner'.
// This must be normalized to 'PRESIDENTE' atomically before the response is returned.
// The partial unique index one_president_per_org guarantees only one PRESIDENTE exists.

import { eq, and, sql } from 'drizzle-orm'
import { auth } from '../../../config/auth'
import { db } from '../../../db'
import { member } from '../../../db/schema'

interface CreateOrgInput {
  name: string
  slug: string
  requestHeaders: HeadersInit
}

export class CreateOrganizationUseCase {
  async execute({ name, slug, requestHeaders }: CreateOrgInput) {
    // 1. Create org via Better Auth (creator becomes 'owner')
    const result = await auth.api.createOrganization({
      body: { name, slug },
      headers: requestHeaders as any,
    })

    if (!result || (result as any).error) {
      throw new Error((result as any)?.error?.message ?? 'Failed to create organization')
    }

    const org = (result as any).data ?? result
    const orgId: string = org.id

    // 2. Normalize creator's role from 'owner' → 'PRESIDENTE' atomically
    // We use raw SQL to update regardless of what Better Auth stored
    await db.execute(sql`
      UPDATE member
      SET role = 'PRESIDENTE'
      WHERE organization_id = ${orgId}
        AND user_id = (
          SELECT user_id FROM member
          WHERE organization_id = ${orgId}
          ORDER BY created_at ASC
          LIMIT 1
        )
    `)

    return org
  }
}

The 401 message should be specific: "Your active organization is no longer accessible. Please switch organizations."
