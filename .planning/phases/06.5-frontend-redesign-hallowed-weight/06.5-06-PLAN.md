Criar `apps/web/src/contexts/event-context.tsx` seguindo exatamente a estrutura de `auth-context.tsx`:

```tsx
import React, { createContext, useContext, useState } from 'react'

const STORAGE_KEY = 'koinonia.selectedEventoId'

interface EventContextType {
  selectedEventoId: string | null
  selectEvento: (id: string) => void
  clearEvento: () => void
}

const EventContext = createContext<EventContextType | undefined>(undefined)

export function EventProvider({ children }: { children: React.ReactNode }) {
  const [selectedEventoId, setSelectedEventoId] = useState<string | null>(
    () => localStorage.getItem(STORAGE_KEY)
  )

  const selectEvento = (id: string) => {
    localStorage.setItem(STORAGE_KEY, id)
    setSelectedEventoId(id)
  }

  const clearEvento = () => {
    localStorage.removeItem(STORAGE_KEY)
    setSelectedEventoId(null)
  }

  return (
    <EventContext.Provider value={{ selectedEventoId, selectEvento, clearEvento }}>
      {children}
    </EventContext.Provider>
  )
}

export function useEventContext() {
  const context = useContext(EventContext)
  if (context === undefined) {
    throw new Error('useEventContext must be used within an EventProvider')
  }
  return context
}
```

- `STORAGE_KEY = 'koinonia.selectedEventoId'` — exatamente esta chave (per RESEARCH.md Pattern 3)
- `clearEvento()` para logout + para quando usuário troca de evento via EventPill
- Inicializar estado a partir de `localStorage.getItem(STORAGE_KEY)` para persistência entre reloads

**Parte 1 — Estender protected-route.tsx:**

Adicionar verificação de `selectedEventoId` após o check de autenticação existente. A rota `/eventos/selecionar` em si NÃO deve ser excluída do redirect (criaria loop) — usar `useLocation` para detectar se já estamos na rota seletora:

```tsx
import { useLocation } from 'react-router-dom'
import { useEventContext } from '../contexts/event-context'

export function ProtectedRoute({ children }: ProtectedRouteProps) {
  const { isAuthenticated, isLoading } = useAuthContext()
  const { selectedEventoId } = useEventContext()
  const location = useLocation()

  if (isLoading) return <LoadingSpinner />
  if (!isAuthenticated) return <Navigate to="/login" replace />

  // Redirecionar para selector apenas se não há evento E não estamos já no selector
  if (!selectedEventoId && location.pathname !== '/eventos/selecionar') {
    return <Navigate to="/eventos/selecionar" replace />
  }

  return <>{children}</>
}
```
