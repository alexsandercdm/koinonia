```tsx
      variant: {
        primary:   'bg-text text-white hover:bg-[#2E2820]',
        secondary: 'bg-transparent text-text border border-border hover:bg-surface-raised',
        gold:      'bg-gold text-white hover:bg-[#A87B2F]',
        ghost:     'bg-transparent text-text-sec hover:bg-[#F0EDE8]',
        danger:    'bg-danger-light text-danger border border-[#F0CACA] hover:bg-[#F5DADA]',
      },
      size: {
        sm: 'h-8 py-1.5 px-3 text-[13px]',
        md: 'h-9 py-2 px-4 text-[14px]',
        lg: 'h-11 py-[11px] px-[22px] text-[15px]',
      },
    },
    defaultVariants: { variant: 'primary', size: 'md' },
  }
)

export interface BtnProps
  extends React.ButtonHTMLAttributes<HTMLButtonElement>,
    VariantProps<typeof btnVariants> {
  icon?: string
  loading?: boolean
}

export const Btn = React.forwardRef<HTMLButtonElement, BtnProps>(
  ({ className, variant, size, icon, loading, children, disabled, ...props }, ref) => {
    return (
      <button
        ref={ref}
        className={cn(btnVariants({ variant, size }), className)}
        disabled={disabled || loading}
        {...props}
      >
        {loading ? (
          <span
            className="material-symbols-outlined leading-none select-none animate-spin"
            style={{ fontSize: 16 }}
          >
            progress_activity
          </span>
        ) : icon ? (
          <span
            className="material-symbols-outlined leading-none select-none"
            style={{ fontSize: 16 }}
          >
            {icon}
          </span>
        ) : null}
        {children}
      </button>
    )
  }
)
Btn.displayName = 'Btn'
```

Criar `FormField.tsx` — wrapper composicional para qualquer campo de formulário:

```tsx
import React from 'react'
import { cn } from '../../lib/utils'

export interface FormFieldProps {
  label?: string
  hint?: string
  error?: string
  required?: boolean
  htmlFor?: string
  className?: string
  children: React.ReactNode
}

export function FormField({
  label,
  hint,
  error,
  required,
  htmlFor,
  className,
  children,
}: FormFieldProps) {
  return (
    <div className={cn('flex flex-col gap-1.5', className)}>
      {label && (
        <label
          htmlFor={htmlFor}
          className="text-[13px] font-medium text-text leading-none"
        >
          {label}
          {required && (
            <span className="ml-0.5 text-danger" aria-hidden="true">
              *
            </span>
          )}
        </label>
      )}
      {children}
      {error ? (
        <p className="text-[11px] text-danger leading-[1.4]" role="alert">
          {error}
        </p>
      ) : hint ? (
        <p className="text-[11px] text-text-ter leading-[1.4]">{hint}</p>
      ) : null}
    </div>
  )
}
```

**Regras de uso:** FormField é composicional — wraps qualquer filho (Input, SelectInput, TextArea, Btn). Não faz validação própria — recebe `error` string de fora (react-hook-form).
