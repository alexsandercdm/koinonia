Criar `Icon.tsx` — primitiva central que substitui lucide-react em toda a aplicação:

```tsx
import React from 'react'
import { cn } from '../../lib/utils'

export interface IconProps {
  name: string
  size?: number
  fill?: boolean
  className?: string
  'aria-hidden'?: boolean | 'true' | 'false'
}

export function Icon({ name, size = 20, fill = false, className, 'aria-hidden': ariaHidden = true }: IconProps) {
  return (
    <span
      className={cn('material-symbols-outlined leading-none select-none', className)}
      style={{
        fontSize: size,
        ...(fill ? { fontVariationSettings: "'FILL' 1, 'wght' 300, 'GRAD' 0, 'opsz' 20" } : {}),
      }}
      aria-hidden={ariaHidden}
    >
      {name}
    </span>
  )
}
```

**Regras de uso:**
- `aria-hidden={true}` por padrão — ícones decorativos. Passar `aria-hidden={false}` quando ícone carrega significado semântico sozinho (sem texto adjacente).
- `fill={true}` para ícones ativos (nav ativo, estado selecionado).
- `size` default 20px — sobrescrever apenas quando spec exige outro tamanho (16px em botões, 40px em EmptyState, etc.).
- NÃO importar lucide em nenhuma hipótese.

**FilterTabs.tsx:**
```tsx
import React from 'react'
import { cn } from '../../lib/utils'

export interface FilterTab {
  id: string
  label: string
  count?: number
}

export interface FilterTabsProps {
  tabs: FilterTab[]
  active: string
  onChange: (id: string) => void
  className?: string
}

export function FilterTabs({ tabs, active, onChange, className }: FilterTabsProps) {
  return (
    <nav role="tablist" aria-label="Filtros" className={cn('flex gap-1', className)}>
      {tabs.map((tab) => {
        const isActive = tab.id === active
        return (
          <button
            key={tab.id}
            role="tab"
            aria-selected={isActive}
            onClick={() => onChange(tab.id)}
            className={cn(
              'inline-flex items-center gap-1 px-3 py-1.5 rounded-pill text-[13px] font-medium transition-all duration-[120ms] ease',
              'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-gold-muted focus-visible:ring-offset-1',
              isActive
                ? 'bg-gold-light text-gold'
                : 'bg-transparent text-text-sec hover:bg-surface-raised'
            )}
          >
            {tab.label}
            {tab.count !== undefined && (
              <span className={cn(
                'text-[11px] tabular-nums',
                isActive ? 'text-gold' : 'text-text-ter'
              )}>
                {tab.count}
              </span>
            )}
          </button>
        )
      })}
    </nav>
  )
}
```

**QRCodeSVG.tsx** (implementação seed-based sem biblioteca — per spec L239):
```tsx
import React from 'react'

export interface QRCodeSVGProps {
  value: string
  size?: number
  className?: string
}

// Geração seed-based determinística para UI mockup/checkin
// Baseada na spec koinonia-primitives.jsx L239-280
function generateQRMatrix(value: string, modules = 21): boolean[][] {
  // Seed a partir do hash da string
  let hash = 0
  for (let i = 0; i < value.length; i++) {
    hash = ((hash << 5) - hash + value.charCodeAt(i)) | 0
  }

  const matrix: boolean[][] = Array.from({ length: modules }, (_, row) =>
    Array.from({ length: modules }, (_, col) => {
      // Finder patterns (cantos)
      const isFinderTL = row < 7 && col < 7
      const isFinderTR = row < 7 && col >= modules - 7
      const isFinderBL = row >= modules - 7 && col < 7
      if (isFinderTL || isFinderTR || isFinderBL) {
        const r = isFinderTL ? row : isFinderTR ? row : row - (modules - 7)
        const c = isFinderTL ? col : isFinderTR ? col - (modules - 7) : col
        const inOuter = r === 0 || r === 6 || c === 0 || c === 6
        const inInner = r >= 2 && r <= 4 && c >= 2 && c <= 4
        return inOuter || inInner
      }
      // Dados seed-based
      const seed = (hash ^ (row * 31 + col * 17)) >>> 0
      return (seed % 3) !== 0
    })
  )
  return matrix
}

export function QRCodeSVG({ value, size = 120, className }: QRCodeSVGProps) {
  const modules = 21
  const matrix = generateQRMatrix(value, modules)
  const cellSize = size / modules

  return (
    <svg
      width={size}
      height={size}
      viewBox={`0 0 ${size} ${size}`}
      className={className}
      aria-label={`QR Code para ${value}`}
      role="img"
    >
      <rect width={size} height={size} fill="white" />
      {matrix.flatMap((row, r) =>
        row.map((cell, c) =>
          cell ? (
            <rect
              key={`${r}-${c}`}
              x={c * cellSize}
              y={r * cellSize}
              width={cellSize}
              height={cellSize}
              fill="#1A1612"
            />
          ) : null
        )
      )}
    </svg>
  )
}
```

**Nota:** QRCodeSVG é visual/decorativo (checkin overlay) — não é QR Code funcional/escaneável. A spec L239 é explícita: "seed-based" e "without lib". Para QR Code real, instalar `qrcode.react` em fase futura.
