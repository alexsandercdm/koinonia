```tsx
import React, { useState, useRef, useEffect } from 'react'
import { useNavigate } from 'react-router-dom'
import { useQuery } from '@tanstack/react-query'
import { cn } from '../../lib/utils'
import { Badge } from '../ui-hw'
import { useEventContext } from '../../contexts/event-context'
import { apiFetch } from '../../lib/api'

interface EventoListItem {
  id: string
  nome: string
  status: 'EM_ANDAMENTO' | 'PLANEJAMENTO' | 'ABERTO' | 'ENCERRADO'
}

const EVENTO_STATUS_BADGE: Record<EventoListItem['status'], { variant: 'success' | 'gold' | 'info' | 'default'; label: string }> = {
  EM_ANDAMENTO: { variant: 'success', label: 'Em andamento' },
  ABERTO: { variant: 'gold', label: 'Aberto' },
  PLANEJAMENTO: { variant: 'info', label: 'Planejamento' },
  ENCERRADO: { variant: 'default', label: 'Encerrado' },
}

export function EventPill() {
  const { selectedEventoId, selectEvento } = useEventContext()
  const [open, setOpen] = useState(false)
  const pillRef = useRef<HTMLDivElement>(null)
  const navigate = useNavigate()

  const { data: eventos = [] } = useQuery({
    queryKey: ['eventos'],
    queryFn: () => apiFetch<EventoListItem[]>('/api/v1/eventos'),
    staleTime: 1000 * 60 * 5,
  })

  // Fechar ao clicar fora
  useEffect(() => {
    function handleClickOutside(e: MouseEvent) {
      if (pillRef.current && !pillRef.current.contains(e.target as Node)) {
        setOpen(false)
      }
    }
    if (open) document.addEventListener('mousedown', handleClickOutside)
    return () => document.removeEventListener('mousedown', handleClickOutside)
  }, [open])

  // Fechar no Esc
  useEffect(() => {
    function handleKeyDown(e: KeyboardEvent) {
      if (e.key === 'Escape') setOpen(false)
    }
    if (open) document.addEventListener('keydown', handleKeyDown)
    return () => document.removeEventListener('keydown', handleKeyDown)
  }, [open])

  const selectedEvento = eventos.find((e) => e.id === selectedEventoId)

  // Retorna null se nenhum evento selecionado
  if (!selectedEventoId || !selectedEvento) return null

  const statusInfo = EVENTO_STATUS_BADGE[selectedEvento.status]

  return (
    <div className="relative" ref={pillRef}>
      {/* Pill button */}
      <button
        onClick={() => setOpen(!open)}
        className={cn(
          'flex items-center gap-1.5 px-[10px] py-[5px] rounded-btn border transition-all duration-[120ms] ease',
          'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-gold-muted focus-visible:ring-offset-2',
          open
            ? 'border-gold-muted bg-gold-light'
            : 'border-border bg-surface-raised hover:border-border-strong'
        )}
        aria-expanded={open}
        aria-haspopup="listbox"
      >
        <span
          className="material-symbols-outlined leading-none text-gold flex-shrink-0"
          style={{ fontSize: 15, fontVariationSettings: "'FILL' 1" }}
        >
          calendar_month
        </span>
        <span className="text-[13px] font-medium text-text max-w-[220px] truncate">
          {selectedEvento.nome}
        </span>
        <Badge variant={statusInfo.variant} size="sm">{statusInfo.label}</Badge>
        <span
          className="material-symbols-outlined leading-none text-text-ter"
          style={{ fontSize: 16 }}
        >
          {open ? 'expand_less' : 'expand_more'}
        </span>
      </button>

      {/* Dropdown */}
      {open && (
        <div
          className="absolute top-[calc(100%+6px)] left-0 min-w-[320px] bg-surface border border-border rounded-card shadow-[0_8px_32px_rgba(0,0,0,0.1)] z-50"
          role="listbox"
        >
          {/* Header */}
          <div className="px-[14px] pt-[10px] pb-2 border-b border-border">
            <p className="text-[11px] font-semibold text-text-ter uppercase tracking-[0.07em]">
              TROCAR EVENTO
            </p>
          </div>

          {/* Event list */}
          <ul className="py-1">
            {eventos.map((evento) => {
              const isSelected = evento.id === selectedEventoId
              const info = EVENTO_STATUS_BADGE[evento.status]
              return (
                <li key={evento.id}>
                  <button
                    role="option"
                    aria-selected={isSelected}
                    onClick={() => {
                      selectEvento(evento.id)
                      setOpen(false)
                    }}
                    className={cn(
                      'w-full flex items-center justify-between gap-2 px-[14px] py-2.5 text-left transition-all duration-[120ms] ease',
                      isSelected
                        ? 'bg-gold-light'
                        : 'hover:bg-surface-raised'
                    )}
                  >
                    <span className={cn(
                      'text-[13px] font-medium truncate',
                      isSelected ? 'text-gold' : 'text-text'
                    )}>
                      {evento.nome}
                    </span>
                    <div className="flex items-center gap-2 flex-shrink-0">
                      <Badge variant={info.variant} size="sm">{info.label}</Badge>
                      {isSelected && (
                        <span className="material-symbols-outlined leading-none text-gold" style={{ fontSize: 14 }}>
                          check
                        </span>
                      )}
                    </div>
                  </button>
                </li>
              )
            })}
          </ul>

          {/* Footer */}
          <div className="px-[14px] pt-2 pb-[10px] border-t border-border">
            <button
              onClick={() => { setOpen(false); navigate('/eventos/selecionar') }}
              className="flex items-center gap-1 text-[13px] text-gold hover:text-[#A87B2F] transition-colors duration-[120ms] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-gold-muted"
            >
              <span className="material-symbols-outlined leading-none" style={{ fontSize: 14 }}>add</span>
              Novo evento
            </button>
          </div>
        </div>
      )}
    </div>
  )
}
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |

| T-06.5-16 | Spoofing | EventPill aria-expanded/haspopup | accept | ARIA attributes apenas indicativos — não bypass de autenticação |
| T-06.5-17 | Denial of Service | EventPill useQuery | accept | staleTime 5min limita frequência de requests. Erros de API silenciosamente resultam em lista vazia (não crash) |
