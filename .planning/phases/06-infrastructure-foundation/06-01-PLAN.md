```typescript
export class ApiError extends Error {
  constructor(
    message: string,
    public readonly status: number,
    public readonly body?: unknown,
  ) {
    super(message)
    this.name = 'ApiError'
  }
}
```

**Change 2 — Replace the generic throw at line 40 (inside the `if (!response.ok)` block):**

Replace:

